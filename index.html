<head>

    <meta charset="UTF-8">

    <meta name="viewport" content="width=device-width, initial-scale=1.0, maximum-scale=1.0, user-scalable=no">

    <title>MAYAMI DRIFT PRO</title>

    <script src="https://cdn.jsdelivr.net/npm/phaser@3.60.0/dist/phaser.min.js"></script>

    <style>body { margin: 0; background: #2ecc71; overflow: hidden; font-family: sans-serif; }</style>

</head>

<body>

<script>

    const config = {

        type: Phaser.AUTO,

        width: window.innerWidth,

        height: window.innerHeight,

        scene: {

            preload: function() {

                this.load.image('car', 'car.png');

            },

            create: function() {

                const {width, height} = this.game.config;

                this.roadWidth = 250;

                this.speed = 5;

                this.score = 0;

                

                // Дорога

                this.add.rectangle(width/2, height/2, this.roadWidth, height * 2, 0x333333);

                

                // UI

                this.scoreText = this.add.text(20, 20, 'SCORE: 0', { fontSize: '24px', fill: '#fff', fontStyle: 'bold' }).setDepth(10);

                

                // Машина и Pivot

                this.car = this.add.sprite(width/2, height * 0.7, 'car').setScale(0.25);

                this.pivot = this.add.circle(0, 0, 15, 0xffd700);

                this.line = this.add.graphics();

                

                this.isDrifting = false;

                this.spawnPivot();

                

                this.input.on('pointerdown', () => this.isDrifting = true);

                this.input.on('pointerup', () => this.isDrifting = false);

            },

            spawnPivot: function() {

                const side = Math.random() > 0.5 ? 1 : -1;

                this.pivot.setPosition(this.game.config.width/2 + (side * 130), this.game.config.height * 0.1);

            },

            update: function() {

                if (this.isDrifting) {

                    Phaser.Math.RotateAround(this.car, this.pivot.x, this.pivot.y, 0.08);

                    this.car.angle += 6;

                    this.line.clear().lineStyle(4, 0xffd700).lineBetween(this.car.x, this.car.y, this.pivot.x, this.pivot.y);

                } else {

                    this.car.y -= this.speed;

                    this.line.clear();

                }


                // Проверка вылета

                const dist = Math.abs(this.car.x - this.game.config.width/2);

                if (dist > this.roadWidth / 2 + 30) {

                    this.scene.restart();

                }


                // Переход уровня

                if (this.car.y < 0) {

                    this.car.y = this.game.config.height;

                    this.score += 10;

                    this.speed += 0.2; // Ускорение игры

                    this.scoreText.setText('SCORE: ' + this.score);

                    this.spawnPivot();

                }

            }

        }

    };

    new Phaser.Game(config);

</script>

</body>
